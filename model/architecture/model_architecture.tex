\section*{Introduction}
This model is a multimodal neural network designed to predict the future 3s trajectory of a vehicle. It processes visual data from two consecutive images with a temporal gap of 0.1s, along with vehicle speed derived from GPS data. The model outputs a sequence of 6 displacement vectors representing the future vehicle's path over 3 seconds.

\section{Model Architecture}


\begin{enumerate}
    \item \textbf{Image Processing Branch}:
    \begin{itemize}
        \item \textbf{Convolutional Block 1}:
        \begin{itemize}
            \item Conv2D: 64 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 64 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item MaxPool2D: $2 \times 2$ 
        \end{itemize}
        
        \item \textbf{Convolutional Block 2}:
        \begin{itemize}
            \item Conv2D: 128 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 128 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item MaxPool2D: $2 \times 2$ 
        \end{itemize}
        
        \item \textbf{Convolutional Block 3}:
        \begin{itemize}
            \item Conv2D: 256 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 256 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 256 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item MaxPool2D: $2 \times 2$ 
        \end{itemize}
        
        \item \textbf{Convolutional Block 4}:
        \begin{itemize}
            \item Conv2D: 512 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 512 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
            \item Conv2D: 512 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item BatchNorm2D
        \end{itemize}
        
        \item \textbf{Spatial Attention}:
        \begin{itemize}
            \item Conv2D: 1 filter, $1 \times 1$ kernel, Sigmoid activation
            \item Attention application through element-wise multiplication
        \end{itemize}
        
        \item \textbf{Feature Concatenation}:
        \begin{itemize}
            \item Concatenate features from both images (creating a 1024-channel feature map)
        \end{itemize}
        
        \item \textbf{Final Convolutional Processing}:
        \begin{itemize}
            \item Conv2D: 1024 filters, $3 \times 3$ kernel, padding=1, ReLU activation
            \item Conv2D: 1024 filters, $1 \times 1$ kernel, ReLU activation
            \item Global Average Pooling to create a 1024-dimensional vector
        \end{itemize}
    \end{itemize}

    \item \textbf{Dynamic Input Processing Branch} (processes vehicle speed):
    \begin{itemize}
        \item Dense layer with 32 units, Tanh activation, input dimension 1
        \item Batch normalization
    \end{itemize}

    \item \textbf{Feature Fusion and Fully Connected Layers}:
    \begin{itemize}
        \item Concatenation of image features (1024) and dynamic features (32)
        \item Dense layer with 2048 units, ReLU activation
        \item Dropout (0.5)
        \item Dense layer with 1024 units, ReLU activation
        \item Dropout (0.5)
        \item Dense layer with 512 units, ReLU activation
        \item Dropout (0.3)
        \item Dense layer with 128 units, ReLU activation
        \item Dense layer with 12 units (output)
        \item Reshape to $(6, 2)$ format - representing 6 vectors with x,y coordinates
    \end{itemize}
\end{enumerate}
